nginx install & Location config:
-------------------------
 sudo apt update
 suso apt nginx -y
 
1.sudo vim /etc/nginx/sites-available/default

server_name yaathvikksy.zapto.org www.yaathvikksy.zapto.org;
location /api/ {
        proxy_pass http://localhost:4000;
		proxy_http_version 1.1;
		proxy_set_header Upgrade $http_upgrade;
		proxy_set_header Connection 'upgrade';
		proxy_set_header Host $host;
		Proxy_cache_bypass $http_upgrade;
}
location / {
    root /var/www/frontend;
    index index.html;
	try_files $uri /index.html;
}

2.sudo nginx -t  ---> verify the config update
3.sudo nginx -s reload --> ngix server reload

setup firwall:

sudo ufw status
sudo ufw status verbose

sudo ufw allow 80/tcp
sudo ufw allow 443/tcp
sudo ufw allow 22/tcp
